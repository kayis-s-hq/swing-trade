package com.swingtrade.data.service;

import com.swingtrade.data.entity.Trade;
import com.swingtrade.data.entity.TradeLabel;
import com.swingtrade.data.repository.TradeLabelRepository;
import com.swingtrade.data.repository.TradeRepository;
import jakarta.transaction.Transactional;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.UUID;

@Service
public class TradeLabelService {

    private static final Logger log = LoggerFactory.getLogger(TradeLabelService.class);

    private final TradeLabelRepository tradeLabelRepository;
    private final TradeRepository tradeRepository;

    public TradeLabelService(
            TradeLabelRepository tradeLabelRepository,
            TradeRepository tradeRepository) {
        this.tradeLabelRepository = tradeLabelRepository;
        this.tradeRepository = tradeRepository;
    }

    @Transactional
    public TradeLabel labelTrade(
            UUID tradeId,
            TradeLabel.ExitReason exitReason,
            String notes,
            Double exitConfidence,
            String labelledBy) {

        Trade trade = tradeRepository.findById(tradeId)
                .orElseThrow(() -> new IllegalArgumentException(
                        "Trade not found with id: " + tradeId));

        if (trade.getStatus() != com.swingtrade.domain.TradeStatus.CLOSED) {
            throw new IllegalStateException(
                    "Only closed trades can be labelled. Trade status: " + trade.getStatus());
        }

        Optional<TradeLabel> existingLabel = tradeLabelRepository.findByTradeId(tradeId);

        if (existingLabel.isPresent()) {
            TradeLabel label = existingLabel.get();
            label.setExitReason(exitReason);
            label.setNotes(notes);
            label.setExitConfidence(java.math.BigDecimal.valueOf(exitConfidence));
            label.setLabelledBy(labelledBy);
            label.setLabelledAt(java.time.LocalDateTime.now());
            log.info("Updated existing label for trade {}: {}", tradeId, exitReason);
            return tradeLabelRepository.save(label);
        }

        TradeLabel label = new TradeLabel();
        label.setTrade(trade);
        label.setExitReason(exitReason);
        label.setNotes(notes);
        label.setExitConfidence(java.math.BigDecimal.valueOf(exitConfidence != null ? exitConfidence : 100.0));
        label.setLabelledBy(labelledBy);
        label.setLabelledAt(java.time.LocalDateTime.now());

        log.info("Created new label for trade {}: {}", tradeId, exitReason);
        return tradeLabelRepository.save(label);
    }

    public Optional<TradeLabel> getLabelByTradeId(UUID tradeId) {
        return tradeLabelRepository.findByTradeId(tradeId);
    }

    public List<TradeLabel> getLabelsByTradeId(UUID tradeId) {
        return tradeLabelRepository.findByTradeId(tradeId);
    }

    public List<TradeLabel> getLabelsByExitReason(TradeLabel.ExitReason reason) {
        return tradeLabelRepository.findByExitReason(reason);
    }

    public Map<TradeLabel.ExitReason, Long> getExitReasonDistribution() {
        return tradeLabelRepository.countByExitReason();
    }

    public List<TradeLabel> getAllLabels() {
        return tradeLabelRepository.findAll();
    }

    @Transactional
    public void deleteLabel(UUID labelId) {
        tradeLabelRepository.deleteById(labelId);
        log.info("Deleted trade label with id: {}", labelId);
    }

    @Transactional
    public void bulkLabelTrades(
            List<UUID> tradeIds,
            TradeLabel.ExitReason exitReason,
            String notes,
            String labelledBy) {

        int count = 0;
        for (UUID tradeId : tradeIds) {
            try {
                labelTrade(tradeId, exitReason, notes, 100.0, labelledBy);
                count++;
            } catch (Exception e) {
                log.warn("Failed to label trade {}: {}", tradeId, e.getMessage());
            }
        }
        log.info("Bulk labelled {} trades with exit reason: {}", count, exitReason);
    }
}

package com.swingtrade.api.controller;

import com.swingtrade.data.entity.TradeLabel;
import com.swingtrade.data.service.TradeLabelService;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

@RestController
@RequestMapping("/api/admin/trades")
public class TradeLabelController {

    private static final Logger log = LoggerFactory.getLogger(TradeLabelController.class);

    private final TradeLabelService tradeLabelService;

    public TradeLabelController(TradeLabelService tradeLabelService) {
        this.tradeLabelService = tradeLabelService;
    }

    @PostMapping("/{tradeId}/label")
    public ResponseEntity<Map<String, Object>> labelTrade(
            @PathVariable UUID tradeId,
            @RequestBody LabelTradeRequest request) {

        try {
            TradeLabel label = tradeLabelService.labelTrade(
                    tradeId,
                    TradeLabel.ExitReason.valueOf(request.getExitReason()),
                    request.getNotes(),
                    request.getExitConfidence(),
                    request.getLabelledBy() != null ? request.getLabelledBy() : "admin");

            Map<String, Object> response = new HashMap<>();
            response.put("success", true);
            response.put("label", mapToDto(label));
            log.info("Successfully labelled trade {} with reason: {}", tradeId, request.getExitReason());
            return ResponseEntity.ok(response);

        } catch (IllegalArgumentException e) {
            Map<String, Object> error = new HashMap<>();
            error.put("success", false);
            error.put("error", "Trade not found: " + e.getMessage());
            return ResponseEntity.badRequest().body(error);

        } catch (IllegalStateException e) {
            Map<String, Object> error = new HashMap<>();
            error.put("success", false);
            error.put("error", e.getMessage());
            return ResponseEntity.badRequest().body(error);

        } catch (IllegalArgumentException e) {
            Map<String, Object> error = new HashMap<>();
            error.put("success", false);
            error.put("error", "Invalid exit reason: " + e.getMessage());
            return ResponseEntity.badRequest().body(error);
        }
    }

    @GetMapping("/{tradeId}/labels")
    public ResponseEntity<Map<String, Object>> getLabels(
            @PathVariable UUID tradeId) {

        List<TradeLabel> labels = tradeLabelService.getLabelsByTradeId(tradeId);

        Map<String, Object> response = new HashMap<>();
        response.put("tradeId", tradeId.toString());
        response.put("labels", labels.stream().map(this::mapToDto).toList());
        response.put("count", labels.size());

        return ResponseEntity.ok(response);
    }

    @GetMapping("/distribution")
    public ResponseEntity<Map<String, Object>> getExitReasonDistribution() {

        Map<TradeLabel.ExitReason, Long> distribution = tradeLabelService.getExitReasonDistribution();

        Map<String, Object> response = new HashMap<>();
        response.put("distribution", distribution);

        long total = distribution.values().stream().mapToLong(Long::longValue).sum();
        response.put("totalLabelled", total);

        return ResponseEntity.ok(response);
    }

    @PostMapping("/bulk-label")
    public ResponseEntity<Map<String, Object>> bulkLabelTrades(
            @RequestBody BulkLabelRequest request) {

        try {
            tradeLabelService.bulkLabelTrades(
                    request.getTradeIds(),
                    TradeLabel.ExitReason.valueOf(request.getExitReason()),
                    request.getNotes(),
                    request.getLabelledBy() != null ? request.getLabelledBy() : "admin");

            Map<String, Object> response = new HashMap<>();
            response.put("success", true);
            response.put("message", "Bulk labelling completed");
            return ResponseEntity.ok(response);

        } catch (IllegalArgumentException e) {
            Map<String, Object> error = new HashMap<>();
            error.put("success", false);
            error.put("error", "Invalid exit reason: " + e.getMessage());
            return ResponseEntity.badRequest().body(error);
        }
    }

    private Map<String, Object> mapToDto(TradeLabel label) {
        Map<String, Object> dto = new HashMap<>();
        dto.put("id", label.getId().toString());
        dto.put("tradeId", label.getTrade().getId().toString());
        dto.put("exitReason", label.getExitReason().name());
        dto.put("exitConfidence", label.getExitConfidence());
        dto.put("notes", label.getNotes());
        dto.put("labelledBy", label.getLabelledBy());
        dto.put("labelledAt", label.getLabelledAt());
        return dto;
    }

    // Request DTOs
    public static class LabelTradeRequest {
        private String exitReason;
        private String notes;
        private Double exitConfidence;
        private String labelledBy;

        public String getExitReason() { return exitReason; }
        public void setExitReason(String exitReason) { this.exitReason = exitReason; }
        public String getNotes() { return notes; }
        public void setNotes(String notes) { this.notes = notes; }
        public Double getExitConfidence() { return exitConfidence; }
        public void setExitConfidence(Double exitConfidence) { this.exitConfidence = exitConfidence; }
        public String getLabelledBy() { return labelledBy; }
        public void setLabelledBy(String labelledBy) { this.labelledBy = labelledBy; }
    }

    public static class BulkLabelRequest {
        private List<UUID> tradeIds;
        private String exitReason;
        private String notes;
        private String labelledBy;

        public List<UUID> getTradeIds() { return tradeIds; }
        public void setTradeIds(List<UUID> tradeIds) { this.tradeIds = tradeIds; }
        public String getExitReason() { return exitReason; }
        public void setExitReason(String exitReason) { this.exitReason = exitReason; }
        public String getNotes() { return notes; }
        public void setNotes(String notes) { this.notes = notes; }
        public String getLabelledBy() { return labelledBy; }
        public void setLabelledBy(String labelledBy) { this.labelledBy = labelledBy; }
    }
}

        double totalWins = trades.stream()

        double totalLosses = trades.stream()
                .filter(Trade::isLoss)
