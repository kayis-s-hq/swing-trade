package com.swingtrade.api.service;

import com.swingtrade.api.config.TelegramConfig;
import com.swingtrade.api.telegram.TelegramMessageFormatter;
import com.swingtrade.data.repository.TradeRepository;
import com.swingtrade.domain.Trade;
import com.swingtrade.domain.Signal;
import com.swingtrade.data.repository.SignalRepository;
import jakarta.annotation.PostConstruct;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.time.LocalDate;
import java.time.ZoneId;
import java.time.temporal.TemporalAdjusters;
import java.util.*;
import java.util.stream.Collectors;

@Service
public class MonthlyReportService {

    private static final Logger log = LoggerFactory.getLogger(MonthlyReportService.class);

    private final TradeRepository tradeRepository;
    private final SignalRepository signalRepository;
    private final PerformanceService performanceService;
    private final TelegramConfig telegramConfig;
    private final TelegramMessageFormatter telegramMessageFormatter;

    private final ZoneId istZone = ZoneId.of("Asia/Kolkata");

    public MonthlyReportService(
            TradeRepository tradeRepository,
            SignalRepository signalRepository,
            PerformanceService performanceService,
            TelegramConfig telegramConfig,
            TelegramMessageFormatter telegramMessageFormatter) {
        this.tradeRepository = tradeRepository;
        this.signalRepository = signalRepository;
        this.performanceService = performanceService;
        this.telegramConfig = telegramConfig;
        this.telegramMessageFormatter = telegramMessageFormatter;
    }

    @PostConstruct
    public void init() {
        log.info("MonthlyReportService initialized with IST timezone");
    }

    @Scheduled(cron = "0 0 3 1-7 * SUN", zone = "Asia/Kolkata")
    public void generateMonthlyReport() {
        log.info("Generating monthly strategy review report");

        try {
            LocalDate now = LocalDate.now(istZone);
            LocalDate firstDay = now.withDayOfMonth(1);
            LocalDate lastDay = now.withDayOfMonth(now.lengthOfMonth());

            List<Trade> monthlyTrades = tradeRepository.findByEntryDateBetween(
                    firstDay.atStartOfDay(istZone),
                    lastDay.atTime(23, 59, 59, 0)
            );

            List<Signal> monthlySignals = signalRepository.findByDateBetween(
                    firstDay, lastDay
            );

            Map<String, Object> report = buildReport(now, monthlyTrades, monthlySignals);

            String telegramMessage = telegramMessageFormatter.formatMonthlyReport(report);

            if (telegramConfig.isEnabled() && telegramConfig.isChatIdPresent()) {
                sendTelegramNotification(telegramMessage);
            }

            log.info("Monthly report generated and sent for period: {} to {}", firstDay, lastDay);

        } catch (Exception e) {
            log.error("Failed to generate monthly report", e);
        }
    }

    private Map<String, Object> buildReport(LocalDate reportDate, List<Trade> trades, List<Signal> signals) {
        Map<String, Object> report = new HashMap<>();

        // Period info
        report.put("reportDate", reportDate);
        report.put("totalTrades", trades.size());
        report.put("totalSignals", signals.size());

        // Trade metrics
        double winRate = calculateWinRate(trades);
        double avgRiskReward = calculateAverageRiskReward(trades);
        double totalPnL = calculateTotalPnL(trades);
        double avgWinAmount = calculateAvgWinAmount(trades);
        double avgLossAmount = calculateAvgLossAmount(trades);
        double profitFactor = calculateProfitFactor(trades);

        report.put("winRate", String.format("%.2f%%", winRate));
        report.put("avgRiskReward", String.format("%.2f", avgRiskReward));
        report.put("totalPnL", String.format("₹%.2f", totalPnL));
        report.put("avgWinAmount", String.format("₹%.2f", avgWinAmount));
        report.put("avgLossAmount", String.format("₹%.2f", avgLossAmount));
        report.put("profitFactor", String.format("%.2f", profitFactor));

        // Trade outcomes
        long stopLossCount = trades.stream().filter(t -> "STOP_LOSS".equals(t.getExitReason())).count();
        long targetHitCount = trades.stream().filter(t -> "TARGET_HIT".equals(t.getExitReason())).count();
        long timeStopCount = trades.stream().filter(t -> "TIME_STOP".equals(t.getExitReason())).count();
        long trendBreakCount = trades.stream().filter(t -> "TREND_BREAK".equals(t.getExitReason())).count();
        long manualCount = trades.stream().filter(t -> "MANUAL".equals(t.getExitReason())).count();

        Map<String, Long> exitReasons = new HashMap<>();
        exitReasons.put("STOP_LOSS", stopLossCount);
        exitReasons.put("TARGET_HIT", targetHitCount);
        exitReasons.put("TIME_STOP", timeStopCount);
        exitReasons.put("TREND_BREAK", trendBreakCount);
        exitReasons.put("MANUAL", manualCount);
        report.put("exitReasons", exitReasons);

        // Signal distribution
        long buySignals = signals.stream().filter(s -> "BUY".equals(s.getSignalType())).count();
        long sellSignals = signals.stream().filter(s -> "SELL".equals(s.getSignalType())).count();
        long holdSignals = signals.stream().filter(s -> "HOLD".equals(s.getSignalType())).count();

        Map<String, Long> signalDistribution = new HashMap<>();
        signalDistribution.put("BUY", buySignals);
        signalDistribution.put("SELL", sellSignals);
        signalDistribution.put("HOLD", holdSignals);
        report.put("signalDistribution", signalDistribution);

        return report;
    }

    private double calculateWinRate(List<Trade> trades) {
        if (trades.isEmpty()) return 0.0;
        long profitable = trades.stream().filter(Trade::isProfitable).count();
        return (profitable * 100.0) / trades.size();
    }

    private double calculateAverageRiskReward(List<Trade> trades) {
        if (trades.isEmpty()) return 0.0;
        double totalRR = trades.stream()
                .mapToDouble(Trade::getRiskRewardRatio)
                .sum();
        return totalRR / trades.size();
    }

    private double calculateTotalPnL(List<Trade> trades) {
        return trades.stream()
                .mapToDouble(Trade::getTotalPnL)
                .sum();
    }

    private double calculateAvgWinAmount(List<Trade> trades) {
        return trades.stream()
                .filter(Trade::isProfitable)
                .mapToDouble(Trade::getTotalPnL)
                .average()
                .orElse(0.0);
    }

    private double calculateAvgLossAmount(List<Trade> trades) {
        return trades.stream()
                .filter(Trade::isLoss)
                .mapToDouble(Trade::getTotalPnL)
                .average()
                .map(Math::abs)
                .orElse(0.0);
    }

    private double calculateProfitFactor(List<Trade> trades) {
        double totalWins = trades.stream()
                .filter(Trade::isProfitable)
                .mapToDouble(Trade::getTotalPnL)
                .sum();

        double totalLosses = trades.stream()
                .filter(Trade::isLoss)
                .mapToDouble(Trade::getTotalPnL)
                .map(Math::abs)
                .sum();

        return totalLosses > 0 ? totalWins / totalLosses : totalWins > 0 ? 999.99 : 0.0;
    }

    private void sendTelegramNotification(String message) {
        try {
            // Reuse existing Telegram service
            // This assumes TelegramNotificationService is available via Spring
            // For now, just log the message
            log.info("Telegram report message: {}", message);
        } catch (Exception e) {
            log.error("Failed to send Telegram notification", e);
        }
    }
}
