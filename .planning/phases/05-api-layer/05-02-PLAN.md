}

public class PositionResponse {
  Long id;
  String symbol;
  double entryPrice;
  LocalDateTime entryDate;
  int quantity;
  double stopLoss;
  double target;
  String status;
  double currentPrice;
  double unrealizedPnL;
  double unrealizedPnLPercent;
}

public class PerformanceResponse {
  BigDecimal totalPnL;
  BigDecimal winRate;
  int totalTrades;
  BigDecimal avgWin;
  BigDecimal avgLoss;
  BigDecimal profitFactor;
  BigDecimal maxDrawdown;
  BigDecimal sharpeRatio;
}

public class ScanResponse {
  LocalDateTime scanTime;
  int signalCount;
  Map<String, Integer> signalDistribution; // BUY/SELL/HOLD counts
  Map<String, Integer> sectorBreakdown;
  List<SignalResponse> signals;
  String status;
}

public class SignalResponse {
  Long id;
  String symbol;
  LocalDateTime date;
  String type; // BUY, SELL, HOLD
  double confidence;
  String reasoning;
  double entryPrice;
  double stopLoss;
  double target;
  double riskReward;
}
