import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;

 * Tests signal generation, scan result processing, and response conversion.

class ScanServiceTest {

    @Mock
    private SignalEngine signalEngine;

    @Mock
    private StockRepository stockRepository;

    @Mock
    private SignalRepository signalRepository;

    @InjectMocks
    private ScanService scanService;

    @BeforeEach
    void setUp() {
        // Reset mocks before each test
    }

    @Test
    @DisplayName("should trigger manual scan and generate signals")
    void testTriggerManualScan() throws Exception {
        // Arrange
        List<String> symbols = List.of("RELIANCE", "TCS", "INFY");
        when(stockRepository.findAll()).thenReturn(List.of());

        // Mock signal generation
        doNothing().when(signalEngine).generateSignalForSymbolNow(any(String.class));

        // Act
        ScanService.ScanResult result = scanService.triggerManualScan();

        // Assert
        assertNotNull(result);
        assertNotNull(result.getScanTime());
        assertEquals("SUCCESS", result.getStatus());

        // Verify signalEngine was called for each symbol
        for (String symbol : symbols) {
            verify(signalEngine, times(1)).generateSignalForSymbolNow(symbol);
        }
    }

    @Test
    @DisplayName("should trigger scan and return ScanResponse")
    void testTriggerScan() {
        // Arrange

        assertNotNull(response.getScanTime());
