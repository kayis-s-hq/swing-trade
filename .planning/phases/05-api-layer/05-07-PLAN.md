```java
public Position createPosition(TradeRequest request) {
    // Validate request
    if (request.getSymbol() == null || request.getQuantity() == null || request.getQuantity() <= 0) {
        throw new IllegalArgumentException("Invalid trade request");
    }

    // Create position with status OPEN
    Position position = new Position();
    position.setSymbol(request.getSymbol());
    position.setEntryPrice(request.getEntryPrice());
    position.setQuantity(request.getQuantity());
    position.setEntryDate(LocalDate.now());
    position.setEntryReason(request.getEntryReason());
    position.setStatus("OPEN");
    position.setCurrentPrice(request.getEntryPrice());

    // Save to repository
    PositionEntity entity = PositionEntity.fromDomain(position);
    positionRepository.save(entity);

    return position;
}
```
